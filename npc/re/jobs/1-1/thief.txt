				mes "[Thief Guide]";
				if (Sex == SEX_MALE)
					mes "Ei, cara.";
				else
					mes "Ei, gatinha~";
				close;
			}
		}else{
			mes "[Thief Guide]";
			if (Sex == SEX_MALE)
				mes "Ei, cara.";

				mes "Ei, gata.";

				mes "[Thief Guildsman]";
				mes "Ei, irmão.";
			}else{
				next;
				mes "[Thief Guildsman]";
				mes "Ei, moça.";
			}
			mes "O que faz aqui? Volte para o seu lugar~ volte~~";

	next;
	mes "[Thief Guildsman]";
	mes "Quer tanto assim ser Gatuno(a)?";
	next;

		mes "[Thief Guildsman]";
		mes "Ah, quer é...? Hum.. bom... a gente vive e aprende vendo gente estranha como você.";

		break;
	case 2:
		mes "[Thief Guildsman]";
		mes "Então por que está aqui? Acha que dá para virar Gatuno(a) assim tão fácil?";
		next;
		break;

		next;
		break;

	next;
	mes "[Thief Guildsman]";
	mes "Então, quer se inscrever para ser Gatuno(a)?";
	next;
	if(select("Sim, quero.:Não.") == 2) {
		mes "[Thief Guildsman]";
		mes "Bom... faça o que quiser~ Siga seu caminho~";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Certo, me diga seu nome. Hã? O quê?";
	mes ""+strcharinfo(0)+". Então "+strcharinfo(0)+", certo...?";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[Thief Guildsman]";
		mes "Vejo sua grande vontade de virar Gatuno(a)......";
		mes "Mas só com vontade você não vence uma luta de verdade, não é?";
		mes "Então vá e chegue pelo menos ao nível 9 da Habilidade Básica.";

	next;
	mes "[Thief Guildsman]";
	mes "Bom, não sou eu que transformo você em Gatuno(a). Eu só recebo as inscrições, entendeu?";
	mes "Se quiser virar Gatuno(a), fale com o cara aqui do lado.";
