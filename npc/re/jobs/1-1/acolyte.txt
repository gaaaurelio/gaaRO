prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Father Mareusis]";
				mes "Ah, sinto que você passou";
				mes "por uma vida passada. Deve ter aprendido muitas coisas antes de entrar em Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Father Mareusis]";
					mes "Infelizmente, acho que você ainda não está pronto(a) para ser Noviço(a). Termine de aprender todas as Habilidades Básicas primeiro.";
					next;
					mes "[Father Mareusis]";
					mes "Enquanto isso,";
					mes "vou esperar até";
					mes "você estar pronto(a).";
					mes "Que Deus esteja";
					mes "com você.";
					close;
				}
				mes "[Father Mareusis]";
				mes "Bem, eu lhe dou as boas-vindas";
				mes "de volta de Valhalla e";
				mes "desejo sorte na sua";
				mes "jornada da nova vida.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Father Mareusis]";
				mes "Agora vá e procure quem precisa da sua ajuda. Que Deus ilumine seu caminho.";
				close;
			}else{
				mes "[Father Mareusis]";
				mes "Agora vá procurar as pessoas que precisam da sua ajuda. Que Deus ilumine seu caminho.";
				close;
			}
		}else{
			mes "[Father Mareusis]";
			mes "Sinto que você passou por uma vida passada. Deve ter aprendido muitas coisas antes de entrar em Valhalla.";
			next;
			mes "[Father Mareusis]";
			mes "Porém, percebo que você não serve para ser Noviço(a). Lembre-se de quem você foi na vida passada e encontre seu caminho.";
			close;
		}
	}
	mes "[Father Mareusis]";
	mes "O que você procura?";
	next;
	switch(select("Mudar de classe para Noviço.:Perguntar os requisitos para ser Noviço.:Desistir.")) {
	case 1:
		mes "[Father Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Você está bem hoje? Pela sua roupa, vejo que você já é Noviço(a). Não está brincando comigo, está?";
			else
				mes "Desculpe, mas parece que você já tem uma classe, não é?";
			close;
		}else{
			mes "Você deseja mesmo se tornar um(a) servo(a) de Deus?";
			mes "Vamos ver se você está pronto(a)... Hum...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Father Mareusis]";
				mes "Oh, céus?! Você ainda não terminou o treinamento básico?! Você tem um longo caminho pela frente! Volte depois de subir seu nível de classe!";
				close;
			}
			mes "[Father Mareusis]";
			mes "Hum... seu nível de classe é suficiente...";
			mes "Ótimo. Agora vou lhe dar a qualificação para se tornar Noviço(a).";
			next;
			mes "[Father Mareusis]";
			mes "Lembre-se sempre de agradecer a Deus, que cuida de nós o tempo todo. No caos e nos momentos difíceis, enfrente as adversidades com fé inabalável.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Father Mareusis]";
			mes "Por fim, quero parabenizá-lo(a) sinceramente por perseverar na sua provação de penitência.";
			close;
		}
	case 2:
		mes "[Father Mareusis]";
		mes "Você deseja se tornar Noviço(a)?";
		mes "Então, cumpra os seguintes requisitos, pensando neles como práticas dadas por Deus.";
		next;
		mes "[Father Mareusis]";
		mes "Primeiro, você precisa chegar pelo menos ao nível de classe 9 de Aprendiz e aprender todas as Habilidades Básicas.";
		mes "Isso é o mais básico, então encare como uma forma de treinar a si mesmo(a).";
		next;
		mes "[Father Mareusis]";
		mes "Quando achar que cumpriu esse requisito, volte a falar comigo. Então você terá uma classe sagrada para espalhar a vontade de Deus.";
		close;
	case 3:
		close;
	}
}
