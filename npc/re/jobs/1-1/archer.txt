	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "Ei, eu conheço você.";
			mes "Você já fez este teste";
			mes "antes, não fez?";
			next;
			mes "[Archer Guildsman]";
			mes "Ah, você deve ter ido";
			mes "a Valhalla e renascido.";
			mes "Uau, que impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Ééé...";
				mes "É melhor aprender todas as Habilidades Básicas antes de se tornar Arqueiro(a).";
				next;
				mes "[Archer Guildsman]";
				mes "Certo, até mais.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Muito bem. Não preciso";
			mes "dizer mais nada.";
			mes "Sei que você será um(a) ótimo(a) Arqueiro(a)...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "Embora não haja recompensa";
			mes "especial desta vez, espero que entenda. Cuide-se.";
			close;
		}else{
			mes "[Archer Guildsman]";
			mes "Oh...?";
			mes "Ei, o que você";
			mes "está fazendo aqui...?";
			next;
			mes "[Archer Guildsman]";
			mes "Percebo que você não nasceu para ser Arqueiro(a). Parece que seu destino é";
			mes "outra coisa...";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "Prazer em conhecê-lo(a). Como posso ajudar?";
	next;
	switch(select("Quero ser Arqueiro(a).:Quero saber os requisitos.:Nada, obrigado(a).")) {
	case 1:
		mes "[Archer Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "Haha, você está brincando comigo...";
			else {
				mes "Sinto muito, mas só Aprendizes podem mudar de classe.";
				mes "Você já tem uma boa classe, não tem?";
			}
			close;
		}else{
			mes "Você é... "+strcharinfo(0)+", certo?";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Bem, você não está no nível de habilidade certo.";
				mes "Seu nível de classe precisa ser pelo menos ^4A4AFF10^000000 e sua Habilidade Básica deve chegar ao ^4A4AFFnível 9";
				next;
				mes "[Archer Guildsman]";
				mes "Um Arqueiro precisa de concentração altíssima, por isso não aceitamos quem tem pouca paciência.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Sua Habilidade Básica já é suficiente..";
			mes "....Hum~~ então você já está pronto(a) para ser Arqueiro(a). Vou cuidar disso agora mesmo.";
			next;
			mes "[Archer Guildsman]";
			mes "Parabéns! Agora você é Arqueiro(a)! Esperamos também que participe ativamente dos programas para reerguer a Guilda dos Arqueiros.";
			mes "Ah, chegaram itens do Departamento de Produção. Tome, pegue estes! São todos seus!";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;	// N_Composite_Bow
			getitem 12004,1;	// Arrow_Container
			getitem 12009,1;	// Silver_Arrow_Container
			getitem 12008,1;	// Fire_Arrow_Container
			next;
			mes "[Archer Guildsman]";
			mes "Com um arco e flechas, agora você é um(a) Arqueiro(a) de verdade.";
			mes "Se abrir a aljava, encontrará flechas dentro e poderá equipá-las.";
			mes "Bem, espero ouvir boas notícias suas. Hora de dizer adeus.";
			mes "Tchau.";
			close;
		}
	case 2:
		mes "[Archer Guildsman]";
		mes "Um Arqueiro tem habilidades com arco e vários talentos.";
		mes "A maior habilidade de um Arqueiro é atacar inimigos de longe.";
		next;
		mes "[Archer Guildsman]";
		mes "Embora tenha menos HP, um Arqueiro consegue atingir inimigos a longa distância,";
		mes "então fica mais seguro numa batalha real.";
		next;
		mes "[Archer Guildsman]";
		mes "Embora o Arqueiro em Ragnarok tenha HP mais baixo, ele tem alta precisão e taxa de ataque, podendo derrotar monstros antes que se aproximem.";
		next;
		mes "[Archer Guildsman]";
		mes "^8C2121Um Arqueiro pode mudar de classe para Caçador.^000000";
		mes "^8C2121Além de Caçador, se for homem, pode virar Bardo e, se for mulher, pode virar Odalisca.^000000";
		close;
	case 3:
		mes "[Archer Guildsman]";
		mes "Se tiver qualquer dúvida, venha me perguntar.";
		close;
	}
}
