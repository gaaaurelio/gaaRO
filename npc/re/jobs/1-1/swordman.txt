izlude_in,74,172,4	script	Swordman#swd	119,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin)) {
			mes "[Swordman]";
			mes "Não...";
			mes "Não pode ser...";
			mes "Você renasceu, não foi?";
			next;
			mes "[Swordman]";
			mes "Vejo que está voltando ao caminho do Espadachim! Depois de se acostumar a empunhar uma espada, não tem mais volta!!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Swordman]";
				mes "Hum? Ah, primeiro você precisa dominar as Habilidades Básicas antes de se tornar Espadachim.";
				next;
				mes "[Swordman]";
				mes "Volte quando terminar de aprender as Habilidades Básicas de Aprendiz.";
				close;
			}
			mes "[Swordman]";
			mes "Excelente! Vou promover você a Espadachim agora mesmo!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Swordman]";
			mes "Hum... Você parece um(a) Espadachim bem experiente. Mesmo assim, tenho certeza de que precisa treinar para melhorar suas habilidades e ganhar força!";
			close;
		}else{
			mes "[Swordman]";
			mes "Hum...?";
			mes "Você é um(a) guerreiro(a)";
			mes "renascido(a), não é?";
			next;
			mes "[Swordman]";
			mes "Hummm...";
			mes "Parece que ser";
			mes "Espadachim não faz parte";
			mes "do seu destino. Sinto muito,";
			mes "mas parece que não há nada";
			mes "que eu possa fazer por você.";
			close;
		}
	}
	mes "[Swordman Guildsman]";
	mes "Esta é a Guilda dos Espadachins.";
	mes "Por que você está aqui?";
	next;
	switch(select("Fale sobre ser Espadachim.:Quero ser Espadachim.:Nada.")) {
	case 1:
		mes "[Swordman Guildsman]";
		mes "Então quer saber mais sobre o poderoso Espadachim? Certo!";
		next;
		mes "[Swordman Guildsman]";
		mes "A característica mais marcante do Espadachim é mostrar sua verdadeira habilidade no combate corpo a corpo.";
		mes "Há três motivos!";
		next;
		mes "[Swordman Guildsman]";
		mes "Primeiro, o Espadachim tem mais HP que as outras classes.";
		mes "Segundo, com exceção de Arcos e Cajados, o Espadachim pode usar todas as outras armas, então luta no seu máximo.";
		mes "E terceiro, a maioria das habilidades do Espadachim causa ataques físicos poderosos.";
		next;
		mes "[Swordman Guildsman]";
		mes "Embora tenha sido uma explicação simples, acredito que você entendeu o essencial do que é ser Espadachim.";
		mes "Na minha opinião, Espadachim é a melhor classe de todas!";
		close;
	case 2:
		mes "[Swordman Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Swordman) {
				mes "Você já é um(a) excelente Espadachim, não é?";
				mes "Dedique-se a ser um(a) grande Espadachim.";
			}else{
				mes "Você já tem outra classe, não tem?";
				mes "Essa brincadeira já foi longe demais.";
			}
			close;
		}else{
			if (!callfunc("F_CanChangeJob")) {
				mes "Sinto dizer, mas para ser Espadachim você precisa chegar pelo menos ao ^4A4AFFnível de classe 10^000000.";
				mes "e ao ^4A4AFFnível 9 da Habilidade Básica^000000.";
				next;
				mes "[Swordman Guildsman]";
				mes "Quer ser Espadachim sem cumprir o requisito mínimo?";
				mes "Acha que ser Espadachim é tão fácil assim?";
				close;
			}
			mes "Hum, seu nível de classe e de Habilidade Básica estão certos.";
			mes "Ótimo. Quer ser Espadachim agora mesmo?";
			next;
			if(select("Sim, quero.:Vou pensar de novo.") == 2) {
				mes "[Swordman Guildsman]";
				mes "Sim. É preciso prudência ao escolher uma classe.";
				mes "Mas sinto muito... que você pense de novo depois de superar todas as dificuldades....";
				close;
			}
			mes "[Swordman Guildsman]";
			mes "Parabéns! De agora em diante, você vai viver a vida de um(a) Espadachim!";
			mes "Vamos fazer isso agora mesmo!";
			next;
			mes "[Swordman Guildsman]";
			mes "Parabéns de novo por se tornar Espadachim, e espero que participe de muitas atividades para reerguer nossa guilda.";
			callfunc "Job_Change",Job_Swordman;
			getitem 13415,1; // N_Falchion
			close;
		}
	case 3:
		close;
	}
}
