//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Kafra Employee]";
		mes "Bem-vindo(a) à";
		mes "Corporação Kafra.";
		mes "Os serviços Kafra";
		mes "estão sempre ao seu lado.";
		mes "Como posso ajudar?";
		break;
	case 1:// Niflheim specific message
		mes "[Kafra Employee]";
		mes "^666666B-beeem-v-viindo";
		mes "à C-coorpoo-r-raçããão";
		mes "K-kaaafrrrra...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Kafra Employee]";
		mes "Bem-vindo(a). ^ff0000" + getguildname( getcharid(2) ) + "^000000 Associado(a).";
		mes "A Corporação Kafra estará com você aonde quer que vá.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Kafra Employee]";
		mes "Então, veio de uma terra distante para estudar nossa cultura ou só está passeando?";
		mes "Em todo caso, por que não fica um pouco?";
		mes "O ar está sempre carregado";
		mes "com o perfume de flores silvestres.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Kafra Employee]";
		mes "Com nossos muitos pontos";
		mes "de serviço Kafra, você nunca";
		mes "está longe de casa.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Usar armazém", "Usar armazém do clã", "Alugar carrinho", "Usar serviço de teleporte", "Cancelar";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Salvar","Usar armazém","Cancelar"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Usar armazém","Cancelar"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Salvar","Usar armazém","Alugar carrinho","Outras informações","Cancelar"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Usar armazém","Alugar carrinho","Outras informações","Cancelar"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Usar armazém","Outras informações","Cancelar"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Salvar","Usar armazém","Alugar carrinho","Cancelar"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Salvar","Usar armazém","Outras informações","Cancelar"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Usar armazém","Alugar carrinho","Usar serviço de teleporte","Outras informações","Cancelar"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Usar armazém","Salvar","Alugar carrinho","Cancelar"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Salvar","Usar armazém","Usar serviço de teleporte","Alugar carrinho","Outras informações","Cancelar"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Salvar")
			return;
		else if (.@K_Menu0$[.@j] == "Usar armazém") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Usar serviço de teleporte") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Kafra Employee]";
				mes "Por causa do ^FF0000Acordo de";
				mes "Transporte Limitado^000000, a";
				mes "Corporação Kafra não pode";
				mes "oferecer serviço de teleporte";
				mes "na República de Schwarzwald.";
				next;
				mes "[Kafra Employee]";
				mes "Pedimos que use";
				mes "o serviço de aeronave";
				mes "no lugar. Agradecemos sua";
				mes "compreensão e cooperação.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Alugar carrinho") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Outras informações") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancelar") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Usar armazém do clã") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas outro membro do clã está usando o armazém do clã";
			mes "agora.  Aguarde até que essa pessoa termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Kafra Employee]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666D-d-desssculpe,";
			mes "v-você é um-um-um";
			mes "Aprendiiiz... P-preciiisa de";
			mes "H-habilidade b-básica nível 6...^000000";
			return;
		}
		mes "Desculpe, mas você";
		mes "precisa da Habilidade";
		mes "Básica do Aprendiz nível 6 para";
		mes "usar o serviço de armazém.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Kafra Employee]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeeny...";
					mes "M-mais z-zeny...!";
					mes "P-preciiiso de 150... zeny...";
					mes "Argh! S-sugando o s-sangue~!^000000";
					return;
				}
				// Standard Message
				mes "Desculpe, mas você não";
				mes "tem zeny suficiente para usar";
				mes "o serviço de armazém. A nossa";
				mes "taxa de acesso ao armazém é de " + .@fee + " zeny.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Kafra Employee]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Obrigada... por... usar...^000000";
		}
		// Normal message
		else {
			mes "[Kafra Employee]";
			mes "Aqui, vou abrir";
			mes "seu armazém para você.";
			mes "Obrigada por usar";
			mes "o serviço Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Kafra Employee]";
	mes "Escolha";
	mes "seu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancelar")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas você não tem";
			mes "zeny suficiente para o serviço";
			mes "de teleporte. A taxa para ir";
			mes "até " + @wrpD$[.@j] + " é de " + @wrpP[.@j] + " zeny.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Farol de Comodo") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mina Abandonada de Mjolnir") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Calabouço Orc") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Kafra Employee]";
		mes "Desculpe, mas o";
		mes "aluguel de carrinho";
		if (checkre(0) == true)
			mes "é exclusivo para a classe Mercador.";
		else {
			mes "é exclusivo para Mercadores,";
			mes "Ferreiros, Mestres-Ferreiros,";
			mes "Alquimistas e Bioquímicos.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Kafra Employee]";
		mes "Você já tem";
		mes "um carrinho equipado.";
		mes "Infelizmente, não podemos";
		mes "alugar mais de um";
		mes "por cliente de cada vez.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Kafra Employee]";
		mes "Você só pode alugar um carrinho depois de aprender a habilidade \"Usar Carrinho\".";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Kafra Employee]";
		mes "A taxa de aluguel";
		mes "do carrinho é de " + .@rental_fee + " zeny. Deseja";
		mes "alugar um carrinho?";
		next;
		if (select("Alugar um carrinho.:Cancelar") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas você";
			mes "não tem zeny suficiente";
			mes "para pagar a taxa de aluguel";
			mes "do carrinho de " + .@rental_fee + " zeny.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Digite a senha do armazém:";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Senha do armazém incorreta.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " tem o orgulho de apresentar um novo serviço:";
		mes "Proteção extra do armazém com senha.";
		next;
		if (select( "Definir nova senha -> 5000z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Seu armazém está protegido por senha. O que deseja fazer?";
	next;
	switch( select( "Trocar a senha -> 5000z", "Remover a senha do armazém -> 1000z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "Primeiro, digite sua ^0000FFsenha antiga^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Senha incorreta. Você não pode definir uma nova senha.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Digite sua senha antes de removê-la.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "A senha não foi removida.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Você não tem zeny suficiente.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "A senha do seu armazém foi removida com sucesso.";
			mes "Obrigada por usar " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Senha incorreta. Não devolveremos seus 1000z.";
			mes "Da próxima vez, digite a senha correta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Agora digite sua ^FF0000nova senha^000000 para proteger seu armazém de ladrões.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "A senha não foi alterada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Você não tem zeny suficiente.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Seu armazém agora está protegido por uma senha secreta.";
	mes "Obrigada por usar " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Digite um número entre 1000~10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Você não pode usar uma senha tão grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Você não deveria usar uma senha tão curta.";
		return 0;
	}
	return .@code_;
}
